import struct
with open('.fizzLocalCache.bin', 'rb') as f:
    data = f.read()
struct_count, char_count = struct.unpack_from('<QQ', data, 0)
structs_start = 16
strings_start = structs_start + (struct_count * 32)
with open('CleanFilenames.txt', 'w') as out:
    for i in range(struct_count):
        offset = structs_start + (i * 32)
        inode, f_type, _, f_offset, f_length = struct.unpack_from('<QB7sQQ', data, offset)
        str_start = strings_start + f_offset
        path = data[str_start:str_start+f_length].decode('utf-8', errors='replace')
        out.write(f'Inode: {inode} | Type: {int(f_type)} | Path: {path}\n')
print('[+] Successfully decoded to CleanFilenames.txt')
